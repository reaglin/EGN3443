5. Hypergeometric Distribution

Example: Electronic Component Sampling Inspection

Problem: A manufacturer receives a batch of 100 electronic components, of which 8 are defective. If a quality control engineer randomly selects 10 components for inspection:

a) What's the probability of finding exactly 2 defective components? b) What's the probability of finding at least 1 defective component? c) How many components should be inspected to have at least a 90% chance of detecting at least one defective component?

Solution in Python (Link)

import numpy as np

import matplotlib.pyplot as plt

from scipy import stats

import math

# Parameters

N = 100  # population size (total components)

K = 8    # number of success states in the population (defective components)

n = 10   # number of draws (sample size)

k = 2    # number of observed successes (defective components found)

# a) Probability of finding exactly 2 defective components

prob_exact_2 = stats.hypergeom.pmf(k=k, M=N, n=K, N=n)

print(f"a) Probability of finding exactly 2 defective components: {prob_exact_2:.4f}")

# b) Probability of finding at least 1 defective component

prob_at_least_1 = 1 - stats.hypergeom.pmf(k=0, M=N, n=K, N=n)

print(f"b) Probability of finding at least 1 defective component: {prob_at_least_1:.4f}")

# c) Finding minimum sample size for 90% detection probability

def find_min_sample_size(N, K, target_prob=0.9):

    n = 1  # Start with the smallest possible sample

    while n <= N:

        # Probability of finding at least 1 defective

        prob = 1 - stats.hypergeom.pmf(k=0, M=N, n=K, N=n)

        if prob >= target_prob:

            return n

        n += 1

    return N  # If no solution found, inspect all components

min_sample_size = find_min_sample_size(N, K, 0.9)

print(f"c) Minimum sample size for 90% detection probability: {min_sample_size} components")

# Parameters

N <- 100  # population size (total components)

K <- 8    # number of success states in the population (defective components)

n <- 10   # number of draws (sample size)

k <- 2    # number of observed successes (defective components found)

# a) Probability of finding exactly 2 defective components

prob_exact_2 <- dhyper(x=k, m=K, n=N-K, k=n)

cat("a) Probability of finding exactly 2 defective components:", round(prob_exact_2, 4), "\n")

# b) Probability of finding at least 1 defective component

prob_at_least_1 <- 1 - dhyper(x=0, m=K, n=N-K, k=n)

cat("b) Probability of finding at least 1 defective component:", round(prob_at_least_1, 4), "\n")

# c) Finding minimum sample size for 90% detection probability

find_min_sample_size <- function(N, K, target_prob=0.9) {

  n <- 1  # Start with the smallest possible sample

  while (n <= N) {

    # Probability of finding at least 1 defective

    prob <- 1 - dhyper(x=0, m=K, n=N-K, k=n)

    if (prob >= target_prob) {

      return(n)

    }

    n <- n + 1

  }

  return(N)  # If no solution found, inspect all components

}

min_sample_size <- find_min_sample_size(N, K, 0.9)

cat("c) Minimum sample size for 90% detection probability:", min_sample_size, "components\n")

# Visualization

# Compare with Binomial Distribution (to illustrate difference)

binom_pmf <- dbinom(x, size=n, prob=K/N)
