4. Poisson Distribution and Applications

Example: Network Traffic Analysis

Problem: In a telecommunications system, data packet arrivals follow a Poisson distribution with an average rate of 5 packets per millisecond.

a) What's the probability of receiving exactly 3 packets in a millisecond? b) What's the probability of receiving more than 10 packets in a millisecond? c) What's the probability of waiting more than 0.4 milliseconds between consecutive packets?

Solution in Python (Link)

import numpy as np

import matplotlib.pyplot as plt

from scipy import stats

# Parameters

lambda_rate = 5  # packets per millisecond

# a) Probability of receiving exactly 3 packets in a millisecond

prob_3_packets = stats.poisson.pmf(k=3, mu=lambda_rate)

print(f"a) Probability of exactly 3 packets in a millisecond: {prob_3_packets:.4f}")

# b) Probability of receiving more than 10 packets in a millisecond

prob_more_than_10 = stats.poisson.sf(k=10, mu=lambda_rate)

print(f"b) Probability of more than 10 packets in a millisecond: {prob_more_than_10:.4f}")

# c) Probability of waiting more than 0.4 milliseconds between consecutive packets

# Time between arrivals follows an exponential distribution with rate = lambda

time_threshold = 0.4  # milliseconds

prob_wait_more_than_0_4 = np.exp(-lambda_rate * time_threshold)

print(f"c) Probability of waiting more than 0.4 ms between packets: {prob_wait_more_than_0_4:.4f}")

# Visualization

plt.figure(figsize=(15, 10))

# Plot the PMF

x = np.arange(0, 20)

pmf = stats.poisson.pmf(x, lambda_rate)

plt.subplot(2, 2, 1)

plt.bar(x, pmf, alpha=0.7)

plt.axvline(x=3, color='red', linestyle='--', alpha=0.7, label='k=3')

plt.xlabel('Number of Packets')

plt.ylabel('Probability')

plt.title(f'Poisson PMF (λ = {lambda_rate})')

plt.grid(alpha=0.3)

plt.legend()

# Plot the CDF and highlight P(X > 10)

x = np.arange(0, 25)

cdf = stats.poisson.cdf(x, lambda_rate)

plt.subplot(2, 2, 2)

plt.step(x, cdf, where='post', alpha=0.7)

plt.axhline(y=stats.poisson.cdf(10, lambda_rate), color='red', linestyle='--', 

            label=f'P(X ≤ 10) = {stats.poisson.cdf(10, lambda_rate):.4f}')

plt.fill_between(x[x > 10], cdf[x > 10], 1, alpha=0.3, step='post', color='green',

                label=f'P(X > 10) = {prob_more_than_10:.4f}')

plt.xlabel('Number of Packets')

plt.ylabel('Cumulative Probability')

plt.title('Poisson CDF')

plt.grid(alpha=0.3)

plt.legend()

# Plot exponential waiting time distribution

t = np.linspace(0, 1, 1000)

exp_pdf = lambda_rate * np.exp(-lambda_rate * t)

plt.subplot(2, 2, 3)

plt.plot(t, exp_pdf)

plt.axvline(x=time_threshold, color='red', linestyle='--', 

            label=f'Time = {time_threshold} ms')

plt.fill_between(t[t >= time_threshold], exp_pdf[t >= time_threshold], alpha=0.3, color='green',

                label=f'P(T > {time_threshold}) = {prob_wait_more_than_0_4:.4f}')

plt.xlabel('Time Between Packets (ms)')

plt.ylabel('Probability Density')

plt.title('Exponential PDF for Packet Inter-arrival Times')

plt.grid(alpha=0.3)

plt.legend()

# Simulate packet arrivals over time

np.random.seed(42)

simulation_time = 10  # milliseconds

timestamps = []

current_time = 0

while current_time < simulation_time:

    # Generate next arrival time (exponentially distributed)

    inter_arrival = np.random.exponential(1/lambda_rate)

    current_time += inter_arrival

    if current_time < simulation_time:

        timestamps.append(current_time)

plt.subplot(2, 2, 4)

plt.eventplot(timestamps, lineoffsets=0, linelengths=0.5, linewidths=2)

plt.xlabel('Time (ms)')

plt.yticks([])

plt.title('Simulated Packet Arrivals Over Time')

plt.xlim(0, simulation_time)

plt.grid(alpha=0.3)

plt.tight_layout()

plt.show()

# Calculate empirical statistics from simulation

inter_arrivals = np.diff(timestamps)

print(f"\nSimulation Results:")

print(f"- Number of packets in {simulation_time} ms: {len(timestamps)}")

print(f"- Average packets per ms: {len(timestamps)/simulation_time:.2f}")

print(f"- Average time between packets: {np.mean(inter_arrivals):.4f} ms")

print(f"- Proportion of inter-arrivals > 0.4 ms: {np.mean(inter_arrivals > time_threshold):.4f}")
