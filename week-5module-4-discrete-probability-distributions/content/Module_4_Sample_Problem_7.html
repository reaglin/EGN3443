7. Combined Distributions in Signal Processing

Example: Digital Communication Channel Analysis

Problem: A digital communication system transmits bits over a noisy channel. The probability of bit error is 0.01. A message consists of 1000 bits with error correction that can handle up to 15 bit errors.

  1. What's the probability that a message will be successfully transmitted?

  2. If we transmit 100 messages, what's the probability that at least 98 will be successful?

  3. What's the distribution of wait time (number of messages) until we encounter the first unsuccessful message?

Solution in Python

import numpy as np

import matplotlib.pyplot as plt

from scipy import stats

import math

# Parameters

p_bit_error = 0.01  # probability of bit error

message_length = 1000  # bits per message

max_errors = 15  # maximum number of bit errors that can be corrected

n_messages = 100  # number of messages transmitted

# 1. Probability of successful message transmission

# This follows a binomial distribution (number of bit errors in a message)

p_message_success = stats.binom.cdf(k=max_errors, n=message_length, p=p_bit_error)

print(f"1. Probability of successful message transmission: {p_message_success:.6f}")

# 2. Probability of at least 98 successful messages out of 100

# This is a binomial distribution of successful messages

p_at_least_98 = stats.binom.sf(k=n_messages-98-1, n=n_messages, p=p_message_success)

print(f"2. Probability of at least 98 successful messages: {p_at_least_98:.6f}")

# 3. Distribution of wait time until first unsuccessful message

# This follows a geometric distribution

p_message_fail = 1 - p_message_success

p_first_fail = stats.geom.pmf(k=np.arange(1, 51), p=p_message_fail)

expected_wait_time = 1 / p_message_fail

print(f"3. Expected wait time until first unsuccessful message: {expected_wait_time:.2f} messages")

# Monte Carlo simulation to verify results

def simulate_message_transmission(p_bit_error, message_length, max_errors):

    # Simulate bits in a message

    errors = np.random.binomial(n=message_length, p=p_bit_error)

    # Check if message is successful

    return errors <= max_errors

def simulate_multiple_messages(p_bit_error, message_length, max_errors, n_messages):

    # Simulate multiple message transmissions

    successes = 0

    for _ in range(n_messages):

        if simulate_message_transmission(p_bit_error, message_length, max_errors):

            successes += 1

    return successes

def simulate_wait_time(p_bit_error, message_length, max_errors):

    # Simulate until first unsuccessful message

    count = 0

    while True:

        count += 1

        if not simulate_message_transmission(p_bit_error, message_length, max_errors):

            return count

# Run simulations

n_simulations = 10000

message_success_count = sum(simulate_message_transmission(p_bit_error, message_length, max_errors) 

                           for _ in range(n_simulations))

p_message_success_sim = message_success_count / n_simulations

multiple_message_results = [simulate_multiple_messages(p_bit_error, message_length, max_errors, n_messages) 

                           for _ in range(n_simulations)]

p_at_least_98_sim = sum(s >= 98 for s in multiple_message_results) / n_simulations

wait_times = [simulate_wait_time(p_bit_error, message_length, max_errors) 

             for _ in range(n_simulations)]

expected_wait_time_sim = np.mean(wait_times)

print("\nSimulation Results:")

print(f"3. Expected wait time: {expected_wait_time_sim:.2f} (vs. analytical {expected_wait_time:.2f})")

# Monte Carlo simulation to verify results

simulate_message_transmission <- function(p_bit_error, message_length, max_errors) {

  # Simulate bits in a message

  errors <- rbinom(n=1, size=message_length, prob=p_bit_error)

  # Check if message is successful

  return(errors <= max_errors)

}

simulate_multiple_messages <- function(p_bit_error, message_length, max_errors, n_messages) {

  # Simulate multiple message transmissions

  successes <- 0

  for (i in 1:n_messages) {

    if (simulate_message_transmission(p_bit_error, message_length, max_errors)) {

      successes <- successes + 1

    }

  }

  return(successes)

}

simulate_wait_time <- function(p_bit_error, message_length, max_errors) {

  # Simulate until first unsuccessful message

  count <- 0

  while (TRUE) {

    count <- count + 1

    if (!simulate_message_transmission(p_bit_error, message_length, max_errors)) {

      return(count)

    }

  }

}
