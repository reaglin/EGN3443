8. Bayesian Decision-Making in Engineering

Example: Fault Detection in Manufacturing

Problem: A manufacturing process has two states: normal operation (80% of the time) and faulty operation (20% of the time). A sensor system detects abnormalities with the following characteristics:

  • When the process is normal, the sensor correctly identifies it 95% of the time (5% false positive)

  • When the process is faulty, the sensor correctly detects it 90% of the time (10% false negative)

If the sensor indicates a fault, what's the probability that the process is actually faulty? How does this probability change as we adjust the prior probability of a fault?

import numpy as np

p_faulty = 0.20  # Prior probability of faulty state

p_normal = 1 - p_faulty  # Prior probability of normal state

p_detect_given_faulty = 0.90  # True positive rate

p_detect_given_normal = 0.05  # False positive rate

# Calculate posterior probability using Bayes' theorem

def bayes_theorem(p_faulty, p_detect_given_faulty, p_detect_given_normal):

    # P(Faulty|Detect) = P(Detect|Faulty) * P(Faulty) / P(Detect)

    # P(Detect) = P(Detect|Faulty) * P(Faulty) + P(Detect|Normal) * P(Normal)

    

    p_normal = 1 - p_faulty

    p_detect = p_detect_given_faulty * p_faulty + p_detect_given_normal * p_normal

    p_faulty_given_detect = (p_detect_given_faulty * p_faulty) / p_detect

    

    return p_faulty_given_detect

# Calculate the posterior probability

p_faulty_given_detect = bayes_theorem(p_faulty, p_detect_given_faulty, p_detect_given_normal)

# Calculate expected cost for different decision policies

# Assume costs: 

# - False positive (stop normal process): $1000

# - False negative (miss fault): $5000

def calculate_expected_cost(decision_threshold, p_faulty=0.2, 

                           cost_false_positive=1000, cost_false_negative=5000):

    """

    Calculate expected cost based on decision threshold.

    - If P(Faulty|Detect) > threshold, we stop the process

    - Otherwise, we continue the process

    """

    prior_range = np.linspace(0.01, 0.99, 100)

    expected_costs = []

    

    for prior in prior_range:

        posterior = bayes_theorem(prior, p_detect_given_faulty, p_detect_given_normal)

        

        # Probability of detection

        p_detect = p_detect_given_faulty * prior + p_detect_given_normal * (1 - prior)

        

        if posterior > decision_threshold:

            # Decision: Stop the process when detection occurs

            # Cost comes from false positives

            cost = cost_false_positive * p_detect_given_normal * (1 - prior)

        else:

            # Decision: Continue the process when detection occurs

            # Cost comes from false negatives

            cost = cost_false_negative * p_detect_given_faulty * prior

        

        expected_costs.append(cost)

    

    return prior_range, expected_costs

# Find the optimal decision threshold for our specific prior

def find_optimal_threshold(p_faulty=0.2, 

                          cost_false_positive=1000, cost_false_negative=5000):

    thresholds = np.linspace(0.01, 0.99, 99)

    costs = []

    

    for threshold in thresholds:

        posterior = bayes_theorem(p_faulty, p_detect_given_faulty, p_detect_given_normal)

        p_detect = p_detect_given_faulty * p_faulty + p_detect_given_normal * (1 - p_faulty)

        

        if posterior > threshold:

            # Decision: Stop the process when detection occurs

            cost = cost_false_positive * p_detect_given_normal * (1 - p_faulty)

        else:

            # Decision: Continue the process when detection occurs

            cost = cost_false_negative * p_detect_given_faulty * p_faulty

        

        costs.append(cost)

    

    optimal_idx = np.argmin(costs)

    return thresholds[optimal_idx], costs[optimal_idx]

calculate_expected_cost <- function(decision_threshold, p_faulty=0.2, 

                                   cost_false_positive=1000, cost_false_negative=5000) {

  # Calculate expected cost based on decision threshold.

  # - If P(Faulty|Detect) > threshold, we stop the process

  # - Otherwise, we continue the process

  

  prior_range <- seq(0.01, 0.99, length.out=100)

  expected_costs <- numeric(length(prior_range))

  

  for (i in 1:length(prior_range)) {

    prior <- prior_range[i]

    posterior <- bayes_theorem(prior, p_detect_given_faulty, p_detect_given_normal)

    

    # Probability of detection

    p_detect <- p_detect_given_faulty * prior + p_detect_given_normal * (1 - prior)

    

    if (posterior > decision_threshold) {

      # Decision: Stop the process when detection occurs

      # Cost comes from false positives

      cost <- cost_false_positive * p_detect_given_normal * (1 - prior)

    } else {

      # Decision: Continue the process when detection occurs

      # Cost comes from false negatives

      cost <- cost_false_negative * p_detect_given_faulty * prior

    }

    

    expected_costs[i] <- cost

  }

  

  return(list(prior_range=prior_range, costs=expected_costs))

}
