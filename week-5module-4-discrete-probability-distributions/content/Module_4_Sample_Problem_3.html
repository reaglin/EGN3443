3. Bernoulli and Binomial Distributions

Example: Semiconductor Chip Testing

Problem: A semiconductor manufacturer produces chips with a 92% yield rate. The company tests each chip before shipping.

a) What's the probability that exactly 45 chips pass out of a batch of 50? b) What's the probability that at least 45 chips pass? c) How many chips should be produced to ensure at least 100 good chips with 99% probability?

Solution in Python (Link)

import numpy as np

import matplotlib.pyplot as plt

from scipy import stats

import math

# Parameters

n = 50  # batch size

p = 0.92  # success probability (yield rate)

k = 45  # number of passing chips

# a) Probability of exactly 45 passing chips

prob_exact = stats.binom.pmf(k=k, n=n, p=p)

print(f"a) Probability of exactly {k} chips passing: {prob_exact:.4f}")

# b) Probability of at least 45 passing chips

prob_at_least = stats.binom.sf(k-1, n, p)  # Survival function (1 - CDF at k-1)

print(f"b) Probability of at least {k} chips passing: {prob_at_least:.4f}")

# c) Finding required production for 99% confidence of at least 100 good chips

def find_production_size(target_good=100, confidence=0.99, yield_rate=0.92):

    # Start with a guess slightly higher than the expected number needed

    n_guess = math.ceil(target_good / yield_rate * 1.1)

    

    while True:

        # Probability of getting at least target_good chips

        prob = stats.binom.sf(target_good - 1, n_guess, yield_rate)

        

        if prob >= confidence:

            # We have enough chips

            return n_guess

        else:

            # Increase guess by 1 and try again

            n_guess += 1

required_production = find_production_size()

print(f"c) Need to produce {required_production} chips to ensure at least 100 good chips with 99% probability")

# Visualization

plt.figure(figsize=(12, 8))

# Plot PMF with highlighting

x = np.arange(0, n+1)

pmf = stats.binom.pmf(x, n, p)

plt.subplot(2, 1, 1)

plt.bar(x, pmf, alpha=0.7)

plt.axvline(x=k, color='red', linestyle='--', alpha=0.7, label=f'k={k}')

plt.fill_between(x[x >= k], pmf[x >= k], alpha=0.3, color='green', label=f'P(X ≥ {k})')

plt.xlabel('Number of Passing Chips')

plt.ylabel('Probability')

plt.title(f'Binomial PMF for n={n}, p={p}')

plt.grid(alpha=0.3)

plt.legend()

# Plot for the required production size problem

plt.subplot(2, 1, 2)

target_good = 100

confidence = 0.99

# Range of production sizes to plot

n_range = np.arange(target_good, required_production + 10)

probs = [stats.binom.sf(target_good - 1, n, p) for n in n_range]

plt.plot(n_range, probs, '-o', markersize=4)

plt.axhline(y=confidence, color='red', linestyle='--', label=f'Confidence = {confidence}')

plt.axvline(x=required_production, color='green', linestyle='--', 

           label=f'Required production = {required_production}')

plt.xlabel('Number of Chips Produced')

plt.ylabel('Probability of ≥ 100 Good Chips')

plt.title('Confidence Level vs. Production Size')

plt.grid(alpha=0.3)

plt.legend()

plt.tight_layout()

plt.show()

# Parameters

n <- 50  # batch size

p <- 0.92  # success probability (yield rate)

k <- 45  # number of passing chips

# a) Probability of exactly 45 passing chips

prob_exact <- dbinom(x=k, size=n, prob=p)

cat("a) Probability of exactly", k, "chips passing:", round(prob_exact, 4), "\n")

# b) Probability of at least 45 passing chips

prob_at_least <- pbinom(q=k-1, size=n, prob=p, lower.tail=FALSE)

cat("b) Probability of at least", k, "chips passing:", round(prob_at_least, 4), "\n")

# c) Finding required production for 99% confidence of at least 100 good chips

find_production_size <- function(target_good=100, confidence=0.99, yield_rate=0.92) {

  # Start with a guess slightly higher than the expected number needed

  n_guess <- ceiling(target_good / yield_rate * 1.1)

  

  while (TRUE) {

    # Probability of getting at least target_good chips

    prob <- pbinom(q=target_good-1, size=n_guess, prob=yield_rate, lower.tail=FALSE)

    

    if (prob >= confidence) {

      # We have enough chips

      return(n_guess)

    } else {

      # Increase guess by 1 and try again

      n_guess <- n_guess + 1

    }

  }

}

required_production <- find_production_size()

cat("c) Need to produce", required_production, "chips to ensure at least 100 good chips with 99% probability\n")

}

text(k*1.2, max(pmf)*0.8, paste("k =", k), pos=4, col="red")

text((n-5)*1.2, max(pmf)*0.6, paste("P(X ≥", k, ") =", round(prob_at_least, 4)), pos=4, col="darkgreen")

target_good <- 100

     main="Confidence Level vs. Production Size",

abline(h=confidence, col="red", lwd=2, lty=2)

abline(v=required_production, col="green", lwd=2, lty=2)

text(required_production, 0.7, paste("Required =", required_production), pos=4, col="darkgreen")

text(required_production-10, confidence+0.01, paste("Confidence =", confidence), pos=4, col="red")
