Example 1: Quality Control in Manufacturing

Imagine a semiconductor manufacturing process where we need to estimate the probability of producing defective chips. We have collected data on 100 randomly sampled chips, where a value of 1 represents a defective chip and 0 represents a non-defective chip. The defect rate for the chips is 15%.

Solution in Python (Link)

import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from scipy.stats import bernoulli

# Example 1: Quality Control in Manufacturing
# Generate some sample data: 100 chips with 15% defect rate
np.random.seed(42)
true_p = 0.15
n_samples = 100
data = bernoulli.rvs(true_p, size=n_samples)

def bernoulli_mle(data):
    """Calculate MLE for Bernoulli parameter p"""
    return np.mean(data)

# Calculate MLE
p_hat = bernoulli_mle(data)
print(f"True p: {true_p}")
print(f"MLE estimate p_hat: {p_hat}")

# Calculate log-likelihood for a range of p values
p_values = np.linspace(0.01, 0.30, 100)
log_likelihoods = []

for p in p_values:
    log_likelihood = np.sum(data * np.log(p) + (1 - data) * np.log(1 - p))
    log_likelihoods.append(log_likelihood)

# Plot log-likelihood function
plt.figure(figsize=(10, 6))
plt.plot(p_values, log_likelihoods)
plt.axvline(p_hat, color='red', linestyle='--', label=f'MLE = {p_hat:.3f}')
plt.axvline(true_p, color='green', linestyle=':', label=f'True p = {true_p}')
plt.xlabel('p (probability of defect)')
plt.ylabel('Log-likelihood')
plt.title('Log-likelihood Function for Bernoulli Distribution')
plt.legend()
plt.grid(True)

# Create confidence interval via profile likelihood
# Find p values where log-likelihood is 1.92 less than max (95% CI based on chi-square)
max_ll = max(log_likelihoods)
threshold = max_ll - 1.92  # Based on chi-square with 1 df (95% CI)

# Find confidence interval bounds
above_threshold = np.array(log_likelihoods) >= threshold
ci_indices = np.where(above_threshold)[0]
ci_lower = p_values[ci_indices[0]]
ci_upper = p_values[ci_indices[-1]]

plt.axhline(threshold, color='blue', linestyle='-.', alpha=0.5, label='95% CI threshold')
plt.fill_between(p_values, min(log_likelihoods), log_likelihoods, 
                 where=(p_values >= ci_lower) & (p_values <= ci_upper),
                 color='gray', alpha=0.3)
plt.legend()

print(f"95% CI: ({ci_lower:.3f}, {ci_upper:.3f})")

plt.figure(figsize=(10, 6))
# Create a bar plot of the observed data (0s and 1s)
values, counts = np.unique(data, return_counts=True)
plt.bar(['Non-defective (0)', 'Defective (1)'], counts, color=['green', 'red'])
plt.axhline(n_samples * (1 - p_hat), color='darkgreen', linestyle='--', 
           label=f'Expected non-defective: {n_samples * (1 - p_hat):.1f}')
plt.axhline(n_samples * p_hat, color='darkred', linestyle='--', 
           label=f'Expected defective: {n_samples * p_hat:.1f}')
plt.ylabel('Count')
plt.title('Observed Chip Quality')
plt.legend()
plt.show()

Solution in R

# Example 1: Quality Control in Manufacturing
set.seed(42)
true_p <- 0.15
n_samples <- 100
data <- rbinom(n_samples, 1, true_p)

# MLE function for Bernoulli
bernoulli_mle <- function(data) {
  return(mean(data))
}

# Calculate MLE
p_hat <- bernoulli_mle(data)
cat("True p:", true_p, "\n")
cat("MLE estimate p_hat:", p_hat, "\n")

# Log-likelihood function for Bernoulli
bernoulli_log_likelihood <- function(p, data) {
  sum(data * log(p) + (1 - data) * log(1 - p))
}

# Calculate log-likelihood for a range of p values
p_values <- seq(0.01, 0.30, length.out = 100)
log_likelihoods <- sapply(p_values, bernoulli_log_likelihood, data = data)

# Plot log-likelihood function
par(mfrow = c(1, 2))
plot(p_values, log_likelihoods, type = "l", 
     xlab = "p (probability of defect)", 
     ylab = "Log-likelihood", 
     main = "Log-likelihood Function for Bernoulli Distribution")
abline(v = p_hat, col = "red", lty = 2)
abline(v = true_p, col = "green", lty = 3)
legend("bottomright", 
       legend = c(paste("MLE =", round(p_hat, 3)), paste("True p =", true_p)),
       col = c("red", "green"), 
       lty = c(2, 3))

# Create confidence interval via profile likelihood
max_ll <- max(log_likelihoods)
threshold <- max_ll - 1.92  # Based on chi-square with 1 df (95% CI)

# Find confidence interval bounds
above_threshold <- log_likelihoods >= threshold
ci_indices <- which(above_threshold)
ci_lower <- p_values[min(ci_indices)]
ci_upper <- p_values[max(ci_indices)]

abline(h = threshold, col = "blue", lty = 4)
polygon(c(p_values[ci_indices], rev(p_values[ci_indices])),
        c(rep(min(log_likelihoods), length(ci_indices)), 
          rev(log_likelihoods[ci_indices])),
        col = rgb(0.5, 0.5, 0.5, 0.3), border = NA)

cat("95% CI: (", round(ci_lower, 3), ", ", round(ci_upper, 3), ")\n", sep = "")

# Bar plot of observed data
counts <- table(factor(data, levels = c(0, 1)))
barplot(counts, names = c("Non-defective (0)", "Defective (1)"), 
        col = c("green", "red"),
        main = "Observed Chip Quality",
        ylab = "Count")
abline(h = n_samples * (1 - p_hat), col = "darkgreen", lty = 2)
abline(h = n_samples * p_hat, col = "darkred", lty = 2)
legend("topright", 
       legend = c(paste("Expected non-defective:", round(n_samples * (1 - p_hat), 1)),
                  paste("Expected defective:", round(n_samples * p_hat, 1))),
       col = c("darkgreen", "darkred"), 
       lty = 2)
