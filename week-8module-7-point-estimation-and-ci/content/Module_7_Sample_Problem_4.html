Module 7 Problem 4: Confidence Interval for Electronic Component Lifetime

Problem Statement

An electrical engineer is testing the lifetime of LED drivers in a lighting system. A random sample of 40 LED drivers is tested until failure, yielding a mean lifetime of 8,500 hours with a sample standard deviation of 450 hours. Calculate the 99% confidence interval for the mean lifetime of these LED drivers.

Step-by-Step Solution

  1. Identify the given values:
    • Sample size (n) = 40
    • Sample mean (x̄) = 8,500 hours
    • Sample standard deviation (s) = 450 hours
    • Confidence level = 99%
  2. Since the population standard deviation is unknown, we use the t-distribution:
    • Degrees of freedom = n - 1 = 39
    • t(0.005, 39) for a 99% confidence level = 2.708
  3. Calculate the margin of error:
    • Margin of error = t(0.005, 39) × (s/√n)
    • Margin of error = 2.708 × (450/√40)
    • Margin of error = 2.708 × (450/6.325)
    • Margin of error = 2.708 × 71.148
    • Margin of error = 192.67
  4. Calculate the confidence interval:
    • CI = x̄ ± margin of error
    • CI = 8,500 ± 192.67
    • CI = (8,307.33, 8,692.67)

Therefore, the 99% confidence interval for the mean lifetime is (8,307.33 hours, 8,692.67 hours).

Solution in Python (link)

import numpy as np
import matplotlib.pyplot as plt
import scipy.stats as stats
import seaborn as sns

# Example 2: LED driver lifetime
sample_mean = 8500  # hours
sample_std = 450    # hours
sample_size = 40
confidence_level = 0.99

# Calculate degrees of freedom and t-critical value
df = sample_size - 1
alpha = 1 - confidence_level
t_critical = stats.t.ppf(1 - alpha/2, df)

# Calculate margin of error
margin_of_error = t_critical * (sample_std / np.sqrt(sample_size))

# Calculate confidence interval
ci_lower = sample_mean - margin_of_error
ci_upper = sample_mean + margin_of_error

print(f"Example 2: LED Driver Lifetime")
print(f"Sample Mean: {sample_mean} hours")
print(f"Sample Size: {sample_size}")
print(f"Sample Standard Deviation: {sample_std} hours")
print(f"Degrees of Freedom: {df}")
print(f"t-critical value (99% confidence): {t_critical:.3f}")
print(f"Margin of Error: {margin_of_error:.2f} hours")
print(f"99% Confidence Interval: ({ci_lower:.2f}, {ci_upper:.2f}) hours")

# Generate a normal distribution for visualization
x = np.linspace(sample_mean - 4*sample_std, sample_mean + 4*sample_std, 1000)
y = stats.norm.pdf(x, sample_mean, sample_std)

# Create visualization
plt.figure(figsize=(10, 6))
plt.plot(x, y, 'b-', linewidth=2, label='Distribution')
plt.axvline(sample_mean, color='red', linestyle='-', linewidth=2, label='Sample Mean')
plt.axvline(ci_lower, color='green', linestyle='--', linewidth=2, label='99% CI Lower')
plt.axvline(ci_upper, color='green', linestyle='--', linewidth=2)

# Shade the confidence interval
ci_x = np.linspace(ci_lower, ci_upper, 100)
ci_y = stats.norm.pdf(ci_x, sample_mean, sample_std)
plt.fill_between(ci_x, ci_y, alpha=0.3, color='green')

plt.title('LED Driver Lifetime with 99% Confidence Interval')
plt.xlabel('Lifetime (hours)')
plt.ylabel('Probability Density')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()

# Create a violin plot to illustrate the distribution
# Generate simulated data based on our sample statistics
np.random.seed(42)
data = np.random.normal(sample_mean, sample_std, 1000)

plt.figure(figsize=(10, 6))
sns.violinplot(y=data, inner="quartile")
plt.axhline(sample_mean, color='red', linestyle='-', linewidth=2, label='Sample Mean')
plt.axhline(ci_lower, color='green', linestyle='--', linewidth=2, label='99% CI')
plt.axhline(ci_upper, color='green', linestyle='--', linewidth=2)
plt.title('Distribution of LED Driver Lifetimes with 99% Confidence Interval')
plt.ylabel('Lifetime (hours)')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()

# Compare different confidence levels
confidence_levels = [0.90, 0.95, 0.99]
margins = []
ci_lowers = []
ci_uppers = []

for cl in confidence_levels:
    alpha = 1 - cl
    t_crit = stats.t.ppf(1 - alpha/2, df)
    margin = t_crit * (sample_std / np.sqrt(sample_size))
    margins.append(margin)
    ci_lowers.append(sample_mean - margin)
    ci_uppers.append(sample_mean + margin)

plt.figure(figsize=(10, 6))
for i, cl in enumerate(confidence_levels):
    plt.plot([cl, cl], [ci_lowers[i], ci_uppers[i]], 'o-', linewidth=2, 
             label=f"{int(cl*100)}% CI: ({ci_lowers[i]:.1f}, {ci_uppers[i]:.1f})")
    
plt.axhline(sample_mean, color='red', linestyle='-', linewidth=2, label='Sample Mean')
plt.title('Comparison of Different Confidence Intervals for LED Driver Lifetime')
plt.xlabel('Confidence Level')
plt.ylabel('Lifetime (hours)')
plt.ylim(sample_mean - 300, sample_mean + 300)
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.show()

# Example 2: LED driver lifetime
sample_mean <- 8500  # hours
sample_std <- 450    # hours
sample_size <- 40
confidence_level <- 0.99

# Calculate degrees of freedom and t-critical value
df <- sample_size - 1
alpha <- 1 - confidence_level
t_critical <- qt(1 - alpha/2, df)

# Calculate margin of error
margin_of_error <- t_critical * (sample_std / sqrt(sample_size))

# Calculate confidence interval
ci_lower <- sample_mean - margin_of_error
ci_upper <- sample_mean + margin_of_error

# Print results
cat("Example 2: LED Driver Lifetime\n")
cat(sprintf("Sample Mean: %g hours\n", sample_mean))
cat(sprintf("Sample Size: %d\n", sample_size))
cat(sprintf("Sample Standard Deviation: %g hours\n", sample_std))
cat(sprintf("Degrees of Freedom: %d\n", df))
cat(sprintf("t-critical value (99%% confidence): %.3f\n", t_critical))
cat(sprintf("Margin of Error: %.2f hours\n", margin_of_error))
cat(sprintf("99%% Confidence Interval: (%.2f, %.2f) hours\n", ci_lower, ci_upper))

# Create necessary libraries
if (!require("ggplot2")) install.packages("ggplot2")
if (!require("gridExtra")) install.packages("gridExtra")
library(ggplot2)
library(gridExtra)

# Generate a normal distribution for visualization
x <- seq(sample_mean - 4*sample_std, sample_mean + 4*sample_std, length.out = 1000)
y <- dnorm(x, mean = sample_mean, sd = sample_std)
df_plot <- data.frame(x = x, y = y)

# Create visualization with ggplot2
p1 <- ggplot(df_plot, aes(x = x, y = y)) +
  geom_line(color = "blue", size = 1) +
  geom_vline(xintercept = sample_mean, color = "red", size = 1) +
  geom_vline(xintercept = ci_lower, color = "green", linetype = "dashed", size = 1) +
  geom_vline(xintercept = ci_upper, color = "green", linetype = "dashed", size = 1) +
  geom_area(data = subset(df_plot, x >= ci_lower & x <= ci_upper), 
            aes(x = x, y = y), fill = "green", alpha = 0.3) +
  labs(title = "LED Driver Lifetime with 99% Confidence Interval",
       x = "Lifetime (hours)",
       y = "Probability Density") +
  theme_minimal() +
  theme(plot.title = element_text(hjust = 0.5))

# Print the plot
print(p1)
ggsave("led_lifetime_ci.png", p1, width = 10, height = 6)

# Compare different confidence levels
confidence_levels <- c(0.90, 0.95, 0.99)
results <- data.frame(
  confidence_level = confidence_levels,
  lower_ci = numeric(length(confidence_levels)),
  upper_ci = numeric(length(confidence_levels))
)

for (i in 1:length(confidence_levels)) {
  cl <- confidence_levels[i]
  alpha <- 1 - cl
  t_crit <- qt(1 - alpha/2, df)
  margin <- t_crit * (sample_std / sqrt(sample_size))
  results$lower_ci[i] <- sample_mean - margin
  results$upper_ci[i] <- sample_mean + margin
}

# Create comparison plot
p3 <- ggplot(results) +

        plot.title = element_text(hjust = 0.5))

# Print the plot
